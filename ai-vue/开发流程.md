## 一、Function Calling（工具调用）能力接入

### 1.1 技术方案选型

| 方案                | 优点                                                           | 缺点                                    |
| ------------------- | -------------------------------------------------------------- | --------------------------------------- |
| **Vercel AI SDK**   | 封装完善、类型安全、流式+工具调用一体化                        | 需引入新依赖，与现有 SSE 流式逻辑有冲突 |
| **原生 JS（推荐）** | 零依赖、完全兼容现有 `useConsultationStream.js` 逻辑、可控性高 | 需手动解析 tool_calls                   |

**决策**：采用**原生 JS 方案**，在现有 DeepSeek SSE 流式对话基础上扩展 Function Calling 解析逻辑，避免引入 Vercel AI SDK 带来的架构冲突。

### 1.2 DeepSeek Function Calling 协议要点

DeepSeek API 兼容 OpenAI 的 Function Calling 协议：

```js
// 请求体中增加 tools 字段
{
  model: "deepseek-chat",
  messages: [...],
  tools: [
    {
      type: "function",
      function: {
        name: "triggerAlert",
        description: "当用户表达自伤、自杀或极度负面情绪时触发预警",
        parameters: {
          type: "object",
          properties: {
            riskLevel: {
              type: "integer",
              enum: [1, 2, 3],
              description: "风险等级：1=关注, 2=预警, 3=危机"
            },
            reason: {
              type: "string",
              description: "触发预警的原因摘要"
            }
          },
          required: ["riskLevel", "reason"]
        }
      }
    },
    {
      type: "function",
      function: {
        name: "searchKnowledgeBase",
        description: "根据用户问题检索心理健康知识库",
        parameters: {
          type: "object",
          properties: {
            query: {
              type: "string",
              description: "检索关键词"
            }
          },
          required: ["query"]
        }
      }
    }
  ],
  tool_choice: "auto"  // auto = 模型自主决定是否调用
}
```

**流式响应中的 tool_calls 解析**：

```js
// SSE 流中，tool_calls 以 delta 增量方式返回
// 第一条 chunk:
{
  choices: [
    {
      delta: {
        tool_calls: [{ index: 0, id: 'call_xxx', type: 'function', function: { name: 'triggerAlert', arguments: '' } }]
      }
    }
  ]
}
// 后续 chunk:
{
  choices: [{ delta: { tool_calls: [{ index: 0, function: { arguments: '{"risk' } }] } }]
}
// ... arguments 逐片拼接
// 最后 chunk:
{
  choices: [
    {
      delta: { tool_calls: [{ index: 0, function: { arguments: 'Level":3,"reason":"..."}' } }] },
      finish_reason: 'tool_calls'
    }
  ]
}
```

### 1.3 开发步骤

#### Step 1：定义工具注册表

新建 `src/composables/useToolCalls.js`：

```js
// 工具注册表 — 所有可被模型调用的函数在此注册
const toolRegistry = {
  triggerAlert: {
    execute: async (args) => {
      const { riskLevel, reason } = args
      return await triggerAlert(riskLevel, reason)
    }
  },
  searchKnowledgeBase: {
    execute: async (args) => {
      const { query } = args
      return await searchKnowledgeBase(query)
    }
  }
}

// 导出 tools 定义（用于 API 请求体）
export const toolDefinitions = [
  {
    type: 'function',
    function: {
      name: 'triggerAlert',
      description: '当用户表达自伤、自杀或极度负面情绪时触发预警',
      parameters: {
        type: 'object',
        properties: {
          riskLevel: { type: 'integer', enum: [1, 2, 3], description: '风险等级：1=关注, 2=预警, 3=危机' },
          reason: { type: 'string', description: '触发预警的原因摘要' }
        },
        required: ['riskLevel', 'reason']
      }
    }
  },
  {
    type: 'function',
    function: {
      name: 'searchKnowledgeBase',
      description: '根据用户问题检索心理健康知识库中的相关文章',
      parameters: {
        type: 'object',
        properties: {
          query: { type: 'string', description: '检索关键词' }
        },
        required: ['query']
      }
    }
  }
]

// 执行工具调用
export async function executeToolCall(toolCall) {
  const { name, arguments: argsStr } = toolCall.function
  const args = JSON.parse(argsStr)
  const tool = toolRegistry[name]
  if (!tool) return { error: `未知工具: ${name}` }
  return await tool.execute(args)
}
```

#### Step 2：改造 `useConsultationStream.js` — 增加 tool_calls 解析

在现有 SSE 流式解析逻辑中，增加对 `tool_calls` 的增量拼接和执行：

```
核心改造点：
1. 请求体增加 tools 字段
2. 流式解析中检测 delta.tool_calls，增量拼接 arguments
3. finish_reason === "tool_calls" 时，执行对应工具函数
4. 将工具执行结果作为 tool 角色消息追加到 messages，再次请求模型获取最终回复
5. 循环处理（模型可能连续调用多个工具），直到 finish_reason !== "tool_calls"
```

**流式处理伪代码**：

```js
async function sendMessage(userMessage) {
  messages.push({ role: "user", content: userMessage })

  let continueLoop = true
  while (continueLoop) {
    const response = await fetch(DEEPSEEK_API_URL, {
      method: "POST",
      headers: { ... },
      body: JSON.stringify({
        model: "deepseek-chat",
        messages,
        tools: toolDefinitions,   // ← 新增
        tool_choice: "auto",      // ← 新增
        stream: true
      })
    })

    // 流式读取
    let toolCallsMap = {}   // index -> { id, name, arguments }
    let finalContent = ""
    let finishReason = null

    for await (const chunk of readSSEStream(response)) {
      const delta = chunk.choices[0].delta
      finishReason = chunk.choices[0].finish_reason

      // 普通文本内容 — 正常流式输出
      if (delta.content) {
        finalContent += delta.content
        appendToCurrentMessage(delta.content)
      }

      // 工具调用 — 增量拼接
      if (delta.tool_calls) {
        for (const tc of delta.tool_calls) {
          const idx = tc.index
          if (!toolCallsMap[idx]) {
            toolCallsMap[idx] = { id: tc.id, name: tc.function.name, arguments: "" }
          }
          if (tc.id) toolCallsMap[idx].id = tc.id
          if (tc.function?.name) toolCallsMap[idx].name = tc.function.name
          if (tc.function?.arguments) toolCallsMap[idx].arguments += tc.function.arguments
        }
      }
    }

    // 处理普通回复
    if (finishReason !== "tool_calls") {
      if (finalContent) messages.push({ role: "assistant", content: finalContent })
      continueLoop = false
      break
    }

    // 处理工具调用
    if (finishReason === "tool_calls") {
      messages.push({
        role: "assistant",
        content: null,
        tool_calls: Object.values(toolCallsMap).map(tc => ({
          id: tc.id, type: "function",
          function: { name: tc.name, arguments: tc.arguments }
        }))
      })

      for (const tc of Object.values(toolCallsMap)) {
        const result = await executeToolCall({
          function: { name: tc.name, arguments: tc.arguments }
        })
        messages.push({ role: "tool", tool_call_id: tc.id, content: JSON.stringify(result) })
      }
    }
  }
}
```

### 3.1 功能描述

### 3.2 技术方案

```
用户上传文件 → 前端解析提取文本 → 拼接到用户消息 → 发送给 DeepSeek
```

**纯前端方案，无需向量数据库**：

- PDF 解析：使用 `pdfjs-dist` 库
- 图片解析：使用 DeepSeek 的 Vision（多模态）能力，直接传图片 base64
- 文本提取后作为上下文注入 system prompt 或用户消息

### 3.3 依赖安装

```bash

```

### 3.4 文件解析模块

新建 `src/composables/useFileParser.js`：

```js
import * as pdfjsLib from 'pdfjs-dist'

// 设置 PDF.js worker
pdfjsLib.GlobalWorkerOptions.workerSrc = new URL('pdfjs-dist/build/pdf.worker.mjs', import.meta.url).toString()

/**
 * 解析 PDF 文件，提取文本内容
 */
export async function parsePDF(file) {
  const arrayBuffer = await file.arrayBuffer()
  const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise
  let fullText = ''

  for (let i = 1; i <= pdf.numPages; i++) {
    const page = await pdf.getPage(i)
    const textContent = await page.getTextContent()
    const pageText = textContent.items.map((item) => item.str).join(' ')
    fullText += `\n--- 第${i}页 ---\n${pageText}`
  }

  return fullText.trim()
}

/**
 * 将图片文件转为 base64（用于 DeepSeek Vision）
 */
export async function imageToBase64(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(reader.result)
    reader.onerror = reject
    reader.readAsDataURL(file)
  })
}

/**
 * 根据文件类型选择解析方式
 */
export async function parseFile(file) {
  const type = file.type

  if (type === 'application/pdf') {
    const text = await parsePDF(file)
    return { type: 'text', content: text }
  }

  if (type.startsWith('image/')) {
    const base64 = await imageToBase64(file)
    return { type: 'image', content: base64 }
  }

  throw new Error('不支持的文件格式，仅支持 PDF 和图片')
}
```

```js
async function searchKnowledgeBase(query) {
  try {
    const res = await fetch(`/api/knowledge/article/search?keyword=${encodeURIComponent(query)}`)
    const data = await res.json()

    if (data.data?.list?.length > 0) {
      const results = data.data.list.slice(0, 3).map((article) => ({
        title: article.title,
        summary: article.summary || article.content?.substring(0, 200)
      }))
      return { found: true, articles: results }
    }

    return { found: false, message: '未找到相关知识库文章' }
  } catch (e) {
    return { found: false, message: '知识库检索失败' }
  }
}
```

### 5.1 安全性

- **API Key 保护**：生产环境必须将 DeepSeek 调用迁移到后端代理 `/api/ai/chat`，前端不应暴露 API Key
- **文件上传安全**：限制文件类型（PDF/JPG/PNG）和大小（≤5MB），防止恶意文件
- **XSS 防护**：解析的文本内容经过 DOMPurify 处理后再渲染

### 5.2 用户体验

- 预警卡片不可过于惊吓，语气温暖关怀，避免"诊断"措辞
- 文件上传过程显示解析进度
- 工具调用期间显示"正在检索知识库..."等状态提示
- 流式输出中 tool_calls 阶段不显示乱码，等待最终文本回复再展示

### 5.3 兼容性

- DeepSeek 的 Function Calling 格式与 OpenAI 兼容，但 `tool_choice` 仅支持 `auto`/`none`/指定函数名
- DeepSeek Vision（图片理解）需使用 `deepseek-chat` 模型，content 传数组格式
- PDF.js Worker 需正确配置，Vite 环境下使用 `import.meta.url` 方式引入
